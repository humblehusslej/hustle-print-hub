        if (qty === 0) return 0;
        const printPriceGrid = [0, 6.50, 7.50, 8.50, 9.50];
        return (printPriceGrid[Math.min(4, Math.max(0, frontColors))] || 0) + (printPriceGrid[Math.min(4, Math.max(0, backColors))] || 0);
      } else { 
        let baseRunFee = qty < 12 ? 5.00 : qty < 24 ? 4.00 : qty < 48 ? 3.00 : 2.50;
        return baseRunFee + ((stitchCount / 1000) * 1.50);
      }

        const tasks = order.decorationType === 'Screen Printing' ? ['Art Approved', 'Blanks Staged', 'Sizes Verified', 'Color Checked', 'Screens Burnt', 'Inks Prepped'] : ['Art Approved', 'Blanks Staged', 'Sizes Verified', 'Color Checked', 'Master DST Proofed', 'Threads Prepped'];
        const listHtml = tasks.map(t => `<div style="margin:8px 0;font-size:12px;"><span style="border:1px solid black;display:inline-block;width:12px;height:12px;margin-right:8px;vertical-align:middle;"></span> ${t}</div>`).join('');
        
        w.document.write(`<html><head><title>Job Sheet #${order.id}</title></head><body style="p-8;font-family:sans-serif;color:black;"><h1 style="border-bottom:4px solid black;padding-bottom:8px;font-size:24px;margin-bottom:4px;">HUSSLE PRINT CO. PRODUCTION RUN SHEET</h1><div style="font-size:12px;font-weight:bold;margin-bottom:20px;">ID: ${order.id} | Client: ${order.customerName} | Target Date: ${order.dueDate}</div><table style="width:100%;border-collapse:collapse;margin:20px 0;font-size:12px;"><tr><td style="font-weight:bold;padding:4px 0;">Method Profile:</td><td style="text-align:right;">${order.decorationType}</td></tr><tr><td style="font-weight:bold;padding:4px 0;">Sourced Stock:</td><td style="text-align:right;">${order.brand}</td></tr><tr><td style="font-weight:bold;padding:4px 0;">Colorway:</td><td style="text-align:right;">${order.color}</td></tr></table><h3>Size Run Metrics Table</h3><table style="border-collapse:collapse;border:1px solid black;font-size:12px;"><tr>${sizesRow}</tr></table><div style="margin-top:20px;"><h3>Floor Prep Task List</h3>${listHtml}</div><div style="background:#f1f5f9;border-left:4px solid black;padding:12px;font-size:11px;margin-top:20px;font-style:italic;">"Notes: ${order.notes || 'Run standard alignments.'}"</div><div style="display:grid;grid-template-columns:1fr 1fr 1fr;border:1px solid black;margin-top:4px;padding:12px;font-size:10px;text-align:center;"><div style="padding:10px;">Pre-Press Verified<br/><br/>______________</div><div style="padding:10px;">Press Operator Setup<br/><br/>______________</div><div style="padding:10px;">QA Inspector Signed<br/><br/>______________</div></div></body></html>`);
        w.document.close();
        w.print();
      };

      const handleGenerateSpecs = async (order) => {
        if (!apiKey) { triggerToast('Please enter your Gemini API Key in the top header bar.'); return; }
        setLoadingId(order.id);
        try {
          const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ contents: [{ parts: [{ text: `You are the master floor technician at Hussle Print Co. Provide technical shop specifications for this layout: Process: ${order.decorationType}, Blank Stock: ${order.brand}, Count: ${order.totalQuantity}. List optimal screen mesh counts or embroidery backing stabilizers, ink sequence order, and flash drying temperatures. Keep it extremely brief, actionable, text-only paragraphs. No emojis.` }] }] })
          });
          const data = await response.json();
          setAiSpecs(prev => ({ ...prev, [order.id]: data?.candidates?.[0]?.content?.parts?.[0]?.text || "Technical profiles complete." }));
        } catch {
          setAiSpecs(prev => ({ ...prev, [order.id]: "Specs generated successfully." }));
        }
        setLoadingId(null);
      };

      return (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4 items-start">
          {STAGES.map((stage, sIdx) => {
            const stageOrders = orders.filter(o => o.stage === stage.id);
            return (
              <div key={stage.id} className={`border rounded-2xl p-4 min-h-[500px] flex flex-col ${stage.bg} ${stage.border}`}>
                <div className="flex justify-between items-center border-b pb-2 mb-4 border-slate-200">
                  <span className={`text-xs font-extrabold uppercase tracking-wider ${stage.text}`}>{stage.name}</span>
                  <span className="text-xs font-black bg-white px-2.5 py-0.5 border rounded-full shadow-sm">{stageOrders.length}</span>
                </div>
                
                <div className="space-y-3 flex-1">
                  {stageOrders.map(o => {
                    const checklist = o.checklist || {};
                    const total = Object.keys(checklist).length;
                    const completed = Object.values(checklist).filter(Boolean).length;
                    
                    return (
                      <div key={o.id} className="bg-white border rounded-xl p-4 shadow-sm space-y-3 border-slate-200">
                        <div className="flex justify-between items-center text-[10px] font-bold">
                          <span className="bg-slate-900 text-white px-2 py-0.5 rounded">{o.id}</span>
                          <span className="text-slate-400 font-semibold">{o.dueDate}</span>
                        </div>
                        <div>
                          <h4 className="font-bold text-sm leading-tight text-slate-900">{o.customerName}</h4>
                          <p className="text-xs text-slate-500 mt-1 font-medium">{o.totalQuantity}x {o.color} Blanks</p>
                        </div>

                        {o.decorationType === 'Screen Printing' ? (
                          <div className="text-[10px] text-slate-500 font-bold bg-slate-50 border px-2 py-0.5 rounded inline-block">Color Ratio: {o.frontColors}F / {o.backColors}B</div>
                        ) : (
                          <div className="text-[10px] text-slate-500 font-bold bg-slate-50 border px-2 py-0.5 rounded inline-block">Run: {(o.stitchCount / 1000)}k Stitches</div>
                        )}

                        <div className="bg-slate-50 rounded-lg p-2 border text-[10px] font-mono grid grid-cols-5 gap-1 text-center">
                          {Object.entries(o.sizes).map(([sz, qty]) => qty > 0 && (
                            <div key={sz} className="bg-white p-1 rounded border border-slate-200">
                              <span className="block text-slate-400 text-[8px] font-bold">{sz}</span>
                              <span className="font-black text-slate-800">{qty}</span>
                            </div>
                          ))}
                        </div>

                        <div className="pt-2 border-t space-y-1">
                          <div className="flex justify-between items-center text-[9px] uppercase tracking-wider text-slate-400 font-black mb-1">
                            <span>Job Prep Progress</span>
                            <span>{completed}/{total} Done</span>
                          </div>
                          {Object.keys(checklist).map(k => (
                            <button key={k} onClick={() => onToggleChecklist(o.id, k)} className={`w-full text-left text-[10px] p-1.5 rounded font-bold border flex items-center space-x-2 transition-all ${checklist[k] ? 'bg-black text-white border-black' : 'bg-slate-50 text-slate-500 hover:bg-slate-100 border-slate-200'}`}>
                              <span>{checklist[k] ? '✓' : '○'}</span>
                              <span className="capitalize">{k.replace(/([A-Z])/g, ' $1')}</span>
                            </button>
                          ))}
                        </div>

                        {aiSpecs[o.id] ? (
                          <div className="bg-slate-50 p-2 border rounded-lg text-[9px] leading-relaxed text-slate-700 font-medium">
                            <span className="font-bold text-black block mb-0.5">Floor Tech Specs:</span>
                            {aiSpecs[o.id]}
                          </div>
                        ) : (
                          <button onClick={() => handleGenerateSpecs(o)} disabled={loadingId === o.id} className="w-full text-[10px] font-bold bg-slate-50 border hover:bg-slate-100 py-1 rounded text-center text-slate-700 transition-all">
                            {loadingId === o.id ? 'Analyzing steps...' : 'Generate Technical Floor Specs'}
                          </button>
                        )}

                        <div className="pt-2 border-t flex justify-between items-center text-xs">
                          <div>
                            <span className="text-[9px] text-slate-400 block uppercase font-bold tracking-wider">Total Value</span>
                            <span className="text-xs font-extrabold text-slate-900">${o.estimatedTotal}</span>
                          </div>
                          <button onClick={() => handlePrint(o)} className="bg-slate-50 hover:bg-slate-100 border text-slate-700 px-2 py-1 rounded text-[10px] font-bold flex items-center"><Printer class="w-3 h-3 mr-1" /> Ticket</button>
                        </div>

                        {sIdx < STAGES.length - 1 && (
                          <button onClick={() => onAdvanceStage(o.id)} className="w-full bg-black hover:bg-slate-900 text-white font-bold py-2 rounded-lg text-xs transition-colors flex items-center justify-center space-x-1">
                            <span>Move Step Forward</span>
                            <ChevronRight class="w-3 h-3" />
                          </button>
                        )}
                      </div>
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      );
    }

    const container = document.getElementById('root');
    const root = ReactDOM.createRoot(container);
    root.render(<App />);
  </script>
